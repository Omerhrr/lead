    59→    59→export async function generateLeadPool(campaignId: string) {
    60→    60→  const campaign = await db.campaign.findUnique({
    61→    61→    where: { id: campaignId },
    62→    62→    include: { project: true },
    63→    63→  })
    64→    64→  if (!campaign) throw new Error("Campaign not found")
    65→    65→
    66→    66→  const filters: LeadFilters = campaign.filters
    67→    67→    ? JSON.parse(campaign.filters)
    68→    68→    : { industries: [], titles: [], geography: [], companySize: [], keywords: [] }
    69→    69→
    70→    70→  const user = await db.user.findUnique({
    71→    71→    where: { id: campaign.project.userId },
    72→    72→    include: { suppress: true },
    73→    73→  })
    74→    74→  const suppressedEmails = new Set(
    75→    75→    (user?.suppress ?? []).filter((s) => s.type === "person").map((s) => s.value.toLowerCase())
    76→    76→  )
    77→    77→  const suppressedCompanies = new Set(
    78→    78→    (user?.suppress ?? [])
    79→    79→      .filter((s) => s.type === "company" || s.type === "domain")
    80→    80→      .map((s) => s.value.toLowerCase())
    81→    81→  )
    82→    82→
    83→    83→  const rows = await db.dbPerson.findMany({ include: { company: true } })
    84→    84→
    85→    85→  const scoreRow = (p: (typeof rows)[number]) => {
    86→    86→    let industryMatch = tokenMatch(p.company.industry, filters.industries)
    87→    87→    let kwMatch = false
    88→    88→    if (!industryMatch && filters.keywords.length) {
    89→    89→      kwMatch = tokenMatch(p.company.industry, filters.keywords) || tokenMatch(p.company.name, filters.keywords)
    90→    90→    }
    91→    91→    const geoMatch = filters.geography.length === 0 || tokenMatch(p.company.country, filters.geography)
    92→    92→    const sizeMatch = filters.companySize.length === 0 || filters.companySize.includes(p.company.size)
    93→    93→    const titleMatch = filters.titles.length === 0 || tokenMatch(p.title, filters.titles)
    94→    94→    const score = Math.min(
    95→    95→      96,
    96→    96→      38 + p.seniority * 9 + (industryMatch ? 10 : kwMatch ? 6 : 0) + (geoMatch ? 6 : 0) + (sizeMatch ? 5 : 0) + (titleMatch ? 6 : 0)
    97→    97→    )
    98→    98→    return { p, industryMatch, kwMatch, geoMatch, sizeMatch, titleMatch, score }
    99→    99→  }
   100→   100→
   101→   101→  const strict = rows.map(scoreRow).filter(
   102→   102→    (r) => r.industryMatch && r.titleMatch && r.geoMatch && r.sizeMatch
   103→   103→  )
   104→   104→  const relaxed = rows.map(scoreRow).filter(
   105→   105→    (r) => (r.industryMatch || r.kwMatch) && r.titleMatch && r.geoMatch
   106→   106→  )
   107→   107→  const fallback = rows.map(scoreRow).filter((r) => r.titleMatch && r.score >= 55)
   108→   108→
   109→   109→  const pool =
   110→   110→    strict.length >= 24 ? strict : relaxed.length >= 24 ? relaxed : [...relaxed, ...fallback]
   111→   111→
   112→   112→  const seen = new Set<string>()
   113→   113→  const chosen = pool
   114→   114→    .sort((a, b) => b.score - a.score)
   115→   115→    .filter((r) => {
   116→   116→      if (seen.has(r.p.email)) return false
   117→   117→      if (suppressedEmails.has(r.p.email.toLowerCase())) return false
   118→   118→      if (suppressedCompanies.has(r.p.company.domain.toLowerCase()) || suppressedCompanies.has(r.p.company.name.toLowerCase())) return false
   119→   119→      seen.add(r.p.email)
   120→   120→      return true
   121→   121→    })
   122→   122→    .slice(0, 32)
   123→   123→
   124→   124→  await db.lead.deleteMany({ where: { campaignId, status: "pool" } })
   125→   125→
   126→   126→  await db.$transaction(
   127→   127→    chosen.map((r) =>
   128→   128→      db.lead.create({
   129→   129→        data: {
   130→   130→          campaignId,
   131→   131→          name: r.p.name,
   132→   132→          title: r.p.title,
   133→   133→          company: r.p.company.name,
   134→   134→          companyDomain: r.p.company.domain,
   135→   135→          industry: r.p.company.industry,
   136→   136→          companySize: r.p.company.size,
   137→   137→          country: r.p.company.country,
   138→   138→          city: r.p.company.city,
   139→   139→          email: r.p.email,
   140→   140→          linkedinUrl: r.p.linkedin,
   141→   141→          score: r.score,
   142→   142→          reasons: r.p.hook,
   143→   143→          status: "pool",
   144→   144→        },
   145→   145→      })
   146→   146→    )
   147→   147→  )
   148→   148→
   149→   149→  await db.campaign.update({
   150→   150→    where: { id: campaignId },
   151→   151→    data: {
   152→   152→      state: "pool_ready",
   153→   153→      nextAction: "Review the lead pool and approve leads to unlock sending",
   154→   154→    },
   155→   155→  })
   156→   156→
   157→   157→  return chosen.length
   158→   158→}

   303→   303→async function advanceCampaign(
   304→   304→  campaign: { id: string; projectId: string; maxTouches: number; delayDays: number; autopilotEnabled: boolean; autoReplyEnabled: boolean; autoReplyDelayMinutes: number; copySubject: string | null; copyBody: string | null; dailyBudgetUsd: number; name: string; offer: string; customerProblem: string; project: { senderName: string; companyName: string } },
   305→   305→  account: { id: string; prepaidBalanceUsd: number; spentTodayUsd: number; spentThisMonthUsd: number; monthlyCapUsd: number; dailyBudgetUsd: number; emailPriceUsd: number; linkedinSeats: number; linkedinSeatBilledMonth: string },
   306→   306→  pulse: { sent: number; replies: number; booked: number; spent: number }
   307→   307→) {
   308→   308→  const now = new Date()
   309→   309→
   310→   310→  // 1) inbound replies due
   311→   311→  const repliesDue = await db.lead.findMany({
   312→   312→    where: {
   313→   313→      campaignId: campaign.id,
   314→   314→      replyDueAt: { lte: now },
   315→   315→      status: "approved",
   316→   316→      OR: [{ convStatus: null }, { convStatus: { in: ["negotiating", "question"] } }],
   317→   317→    },
   318→   318→  })
   319→   319→  for (const lead of repliesDue) {
   320→   320→    await processInbound(campaign, lead, account)
   321→   321→    pulse.replies++
   322→   322→  }
   323→   323→
   324→   324→  // 2) autopilot actions due
   325→   325→  if (campaign.autopilotEnabled && campaign.autoReplyEnabled) {
   326→   326→    const agentDue = await db.lead.findMany({
   327→   327→      where: { campaignId: campaign.id, agentDueAt: { lte: now }, agentStage: { not: null }, convStatus: { in: ["interested", "negotiating", "question", "propose"] } },
   328→   328→    })
   329→   329→    for (const lead of agentDue.slice(0, 3)) {
   330→   330→      await processAutopilot(campaign, lead)
   331→   331→    }
   332→   332→  }
   333→   333→
   334→   334→  // 3) budget guards
   335→   335→  const capacity = await campaignCapacity(campaign.id, campaign.projectId)
   336→   336→  const quota = Math.max(2, Math.ceil(capacity / 12))
   337→   337→  const perSendCost = account.emailPriceUsd || EMAIL_PRICE_USD
   338→   338→  const balanceOk = account.prepaidBalanceUsd >= perSendCost
   339→   339→  const dailyOk = account.spentTodayUsd + perSendCost <= Math.min(account.dailyBudgetUsd, campaign.dailyBudgetUsd)
   340→   340→  const monthlyOk = account.spentThisMonthUsd < account.monthlyCapUsd
   341→   341→
   342→   342→  if (!balanceOk) {
   343→   343→    await db.campaign.update({
   344→   344→      where: { id: campaign.id },
   345→   345→      data: { nextAction: "Out of credits — top up billing to resume outreach" },
   346→   346→    })
   347→   347→    return
   348→   348→  }
   349→   349→  if (!monthlyOk) {
   350→   350→    await db.campaign.update({
   351→   351→      where: { id: campaign.id },
   352→   352→      data: { nextAction: "Monthly cap reached — raise the cap in billing" },
   353→   353→    })
   354→   354→    return
   355→   355→  }
   356→   356→  if (!dailyOk) {
   357→   357→    await db.campaign.update({
   358→   358→      where: { id: campaign.id },
   359→   359→      data: { nextAction: `Daily budget reached (${campaign.dailyBudgetUsd} USD) — resumes next cycle` },
   360→   360→    })
   361→   361→    return
   362→   362→  }
   363→   363→  if (capacity === 0) {
   364→   364→    await db.campaign.update({
   365→   365→      where: { id: campaign.id },
   366→   366→      data: { nextAction: "No active sending domains — buy domains in the Domains tab" },
   367→   367→    })
   368→   368→    return
   369→   369→  }
   370→   370→
   371→   371→  // 4) sends (first touches + follow-ups)
   372→   372→  const eligible = await db.lead.findMany({
   373→   373→    where: {
   374→   374→      campaignId: campaign.id,
   375→   375→      status: "approved",
   376→   376→      convStatus: null,
   377→   377→      OR: [
   378→   378→        { touchCount: 0 },
   379→   379→        { nextTouchAt: { lte: now }, touchCount: { lt: campaign.maxTouches } },
   380→   380→      ],
   381→   381→    },
   382→   382→    orderBy: [{ touchCount: "asc" }, { score: "desc" }],
   383→   383→    take: quota,
   384→   384→  })
   385→   385→
   386→   386→  let emails = 0
   387→   387→  let linkedin = 0
   388→   388→  for (const lead of eligible) {
   389→   389→    const isFollowup = lead.touchCount > 0
   390→   390→    const seatDue = account.linkedinSeatBilledMonth !== monthStr()
   391→   391→    const seatCost = account.linkedinSeats * LINKEDIN_SEAT_PRICE_USD
   392→   392→    const seatAffordable = !seatDue || account.prepaidBalanceUsd >= seatCost + perSendCost
   393→   393→    const useLinkedin = !!lead.linkedinUrl && lead.touchCount >= 1 && seatAffordable && Math.random() < 0.3
   394→   394→    const channel = useLinkedin ? "linkedin" : "email"
   395→   395→    const touchIndex = lead.touchCount + 1
   396→   396→
   397→   397→    const subject = isFollowup
   398→   398→      ? touchIndex === 2
   399→   399→        ? `Re: ${(campaign.copySubject || "Quick question").replace(/^Re:\s*/i, "")}`
   400→   400→        : `Following up (last one, promise)`
   401→   401→      : (campaign.copySubject || "Quick question")
   402→   402→    const bodyTpl = campaign.copyBody || `Hi {{first_name}}, quick idea for {{company}}.`
   403→   403→    let body = fillCopy(bodyTpl, lead, campaign.project.senderName)
   404→   404→    if (isFollowup) {
   405→   405→      body = `Hi ${lead.name.split(" ")[0]}, just floating this back to the top of your inbox — happy to share how similar teams fixed "${lead.reasons || "this"}" with us.\n\nWorth 15 minutes?`
   406→   406→      if (touchIndex >= campaign.maxTouches) body += `\n\nI'll close the loop on my end either way — good luck with ${lead.company}!`
   407→   407→    }
   408→   408→
   409→   409→    await db.message.create({
   410→   410→      data: {
   411→   411→        campaignId: campaign.id,
   412→   412→        leadId: lead.id,
   413→   413→        direction: "out",
   414→   414→        channel,
   415→   415→        kind: "touch",
   416→   416→        touchIndex,
   417→   417→        subject: channel === "email" ? subject : null,
   418→   418→        body,
   419→   419→        status: Math.random() < 0.03 ? "bounced" : "sent",
   420→   420→      },
   421→   421→    })
   422→   422→
   423→   423→    const nextTouchAt =
   424→   424→      touchIndex < campaign.maxTouches
   425→   425→        ? new Date(now.getTime() + campaign.delayDays * FOLLOWUP_MS_PER_DAY)
   426→   426→        : null
   427→   427→
   428→   428→    const willReply =
   429→   429→      lead.replyDueAt === null &&
   430→   430→      Math.random() < (REPLY_CHANCE_BY_TOUCH[Math.min(touchIndex - 1, REPLY_CHANCE_BY_TOUCH.length - 1)] ?? 0.08)
   431→   431→
   432→   432→    await db.lead.update({
   433→   433→      where: { id: lead.id },
   434→   434→      data: {
   435→   435→        touchCount: touchIndex,
   436→   436→        nextTouchAt,
   437→   437→        replyDueAt: willReply ? new Date(now.getTime() + rand(15000, 55000)) : lead.replyDueAt,
   438→   438→        replyChannel: willReply ? "email" : lead.replyChannel,
   439→   439→      },
   440→   440→    })
   441→   441→
   442→   442→    if (channel === "email") emails++
   443→   443→    else linkedin++
   444→   444→  }
   445→   445→
   446→   446→  if (emails + linkedin > 0) {
   447→   447→    const cost = emails * perSendCost + linkedin * LINKEDIN_TOUCH_PRICE_USD
   448→   448→    let description = `${campaign.name}: ${emails} email${emails === 1 ? "" : "s"} sent`
   449→   449→    if (linkedin > 0) description += `, ${linkedin} LinkedIn touch${linkedin === 1 ? "" : "es"}`
   450→   450→    await charge(account.id, cost, description, { emails, linkedin })
   451→   451→    pulse.sent += emails
   452→   452→    pulse.spent += cost
   453→   453→
   454→   454→    // LinkedIn seat billing: once per month on first touch — never overdraw:
   455→   455→    // if the balance can't cover the seats, LinkedIn touches fall back to email.
   456→   456→    if (linkedin > 0 && account.linkedinSeatBilledMonth !== monthStr()) {
   457→   457→      const seatCost = account.linkedinSeats * LINKEDIN_SEAT_PRICE_USD
   458→   458→      const fresh = await db.billingAccount.findUnique({ where: { id: account.id } })
   459→   459→      if (fresh && fresh.prepaidBalanceUsd >= seatCost) {
   460→   460→        await db.billingAccount.update({
   461→   461→          where: { id: account.id },
   462→   462→          data: {
   463→   463→            prepaidBalanceUsd: fresh.prepaidBalanceUsd - seatCost,
   464→   464→            linkedinSeatBilledMonth: monthStr(),
   465→   465→          },
   466→   466→        })
   467→   467→        await db.creditTransaction.create({
   468→   468→          data: {
   469→   469→            accountId: account.id,
   470→   470→            type: "linkedin_seats",
   471→   471→            amountUsd: -seatCost,
   472→   472→            balanceAfter: fresh.prepaidBalanceUsd - seatCost,
   473→   473→            description: `LinkedIn seats — ${account.linkedinSeats} seat${account.linkedinSeats === 1 ? "" : "s"} (${monthStr()})`,
   474→   474→          },
   475→   475→        })
   476→   476→      } else if (fresh) {
   477→   477→        await db.campaign.update({
   478→   478→          where: { id: campaign.id },
   479→   479→          data: {
   480→   480→            nextAction: `LinkedIn seats not charged — balance below ${seatCost} USD; touches sent via email instead`,
   481→   481→          },
   482→   482→        })
   483→   483→      }
   484→   484→    }
   485→   485→  }
   486→   486→
   487→   487→  // 5) completion check
   488→   488→  const approvedLeads = await db.lead.findMany({ where: { campaignId: campaign.id, status: "approved" } })
   489→   489→  const done = approvedLeads.every(
   490→   490→    (l) =>
   491→   491→      ["booked", "closed", "unsubscribed", "active"].includes(l.convStatus ?? "") ||
   492→   492→      (l.convStatus === null && l.touchCount >= campaign.maxTouches && !l.nextTouchAt)
   493→   493→  )
   494→   494→  if (approvedLeads.length > 0 && done) {
   495→   495→    await db.campaign.update({
   496→   496→      where: { id: campaign.id },
   497→   497→      data: { state: "completed", sendingNow: false, nextAction: "Campaign completed — review hot leads in the inbox" },
   498→   498→    })
   499→   499→  } else {
   500→   500→    await db.campaign.update({
   501→   501→      where: { id: campaign.id },
   502→   502→      data: { nextAction: "Sending — sequences running on schedule" },
   503→   503→    })
   504→   504→  }
   505→   505→}

   509→   509→async function processInbound(
   510→   510→  campaign: { id: string; autopilotEnabled: boolean; autoReplyEnabled: boolean; autoReplyDelayMinutes: number },
   511→   511→  lead: { id: string; name: string; company: string; convStatus: string | null; replyChannel: string | null },
   512→   512→  account: { id: string }
   513→   513→) {
   514→   514→  const now = new Date()
   515→   515→  let outcome: string
   516→   516→  if (lead.convStatus === "negotiating") {
   517→   517→    outcome = "negotiating"
   518→   518→  } else if (lead.convStatus === "question") {
   519→   519→    outcome = Math.random() < 0.5 ? "interested" : "not_interested"
   520→   520→  } else {
   521→   521→    outcome = pickWeighted(FIRST_REPLY_OUTCOMES)
   522→   522→  }
   523→   523→
   524→   524→  const pool = INBOUND_TEXT[outcome] ?? INBOUND_TEXT.not_interested
   525→   525→  const text = pool[randInt(0, pool.length - 1)]
   526→   526→
   527→   527→  await db.message.create({
   528→   528→    data: {
   529→   529→      campaignId: campaign.id,
   530→   530→      leadId: lead.id,
   531→   531→      direction: "in",
   532→   532→      channel: lead.replyChannel || "email",
   533→   533→      kind: "reply",
   534→   534→      body: text,
   535→   535→      status: "sent",
   536→   536→    },
   537→   537→  })
   538→   538→
   539→   539→  const hot = outcome === "interested" || outcome === "negotiating"
   540→   540→  const updates: Record<string, unknown> = { replyDueAt: null, sentiment: outcome }
   541→   541→  switch (outcome) {
   542→   542→    case "interested":
   543→   543→      updates.convStatus = "interested"
   544→   544→      updates.isHot = true
   545→   545→      if (campaign.autopilotEnabled && campaign.autoReplyEnabled) {
   546→   546→        updates.agentDueAt = new Date(now.getTime() + Math.max(10, campaign.autoReplyDelayMinutes) * 1000)
   547→   547→        updates.agentStage = "propose_meeting"
   548→   548→      } else {
   549→   549→        updates.convStatus = "active"
   550→   550→      }
   551→   551→      break
   552→   552→    case "question":
   553→   553→      updates.convStatus = "question"
   554→   554→      if (campaign.autopilotEnabled && campaign.autoReplyEnabled) {
   555→   555→        updates.agentDueAt = new Date(now.getTime() + Math.max(10, campaign.autoReplyDelayMinutes) * 1000)
   556→   556→        updates.agentStage = "answer"
   557→   557→      } else {
   558→   558→        updates.convStatus = "active"
   559→   559→      }
   560→   560→      break
   561→   561→    case "not_interested":
   562→   562→      updates.convStatus = "closed"
   563→   563→      break
   564→   564→    case "unsubscribe":
   565→   565→      updates.convStatus = "unsubscribed"
   566→   566→      updates.status = "suppressed"
   567→   567→      break
   568→   568→    case "negotiating":
   569→   569→      // lead accepted the meeting
   570→   570→      updates.convStatus = "negotiating"
   571→   571→      updates.isHot = true
   572→   572→      updates.agentDueAt = new Date(now.getTime() + Math.max(10, campaign.autoReplyDelayMinutes) * 1000)
   573→   573→      updates.agentStage = "confirm"
   574→   574→      updates.meetingAt = new Date(now.getTime() + MEETING_LEAD_DAYS[randInt(0, 3)] * 24 * 3600 * 1000)
   575→   575→      break
   576→   576→    case "booked_confirm":
   577→   577→      updates.convStatus = "booked"
   578→   578→      break
   579→   579→  }
   580→   580→
   581→   581→  await db.lead.update({ where: { id: lead.id }, data: updates })
   582→   582→  await noteReplyEvent(account.id, hot)
   583→   583→}

   647→   647→export async function campaignAction(campaignId: string, action: string) {
   648→   648→  const campaign = await db.campaign.findUnique({ where: { id: campaignId } })
   649→   649→  if (!campaign) throw new Error("Campaign not found")
   650→   650→
   651→   651→  switch (action) {
   652→   652→    case "approve":
   653→   653→      await db.campaign.update({
   654→   654→        where: { id: campaignId },
   655→   655→        data: { state: "approved", nextAction: "Press Start to begin outreach" },
   656→   656→      })
   657→   657→      break
   658→   658→    case "start": {
   659→   659→      const owned = await db.sendingDomain.count({ where: { projectId: campaign.projectId, status: "active" } })
   660→   660→      if (owned === 0) {
   661→   661→        return { ok: false, error: "No active sending domains. Buy domains in the Domains tab first." }
   662→   662→      }
   663→   663→      await db.campaign.update({
   664→   664→        where: { id: campaignId },
   665→   665→        data: { state: "sending", sendingNow: true, nextAction: "Sending — sequences running on schedule" },
   666→   666→      })
   667→   667→      break
   668→   668→    }
   669→   669→    case "stop":
   670→   670→      await db.campaign.update({
   671→   671→        where: { id: campaignId },
   672→   672→        data: { sendingNow: false, state: "paused", nextAction: "Paused — press Resume to continue" },
   673→   673→      })
   674→   674→      break
   675→   675→    case "resume":
   676→   676→      await db.campaign.update({
   677→   677→        where: { id: campaignId },
   678→   678→        data: { sendingNow: true, state: "sending", nextAction: "Sending — sequences running on schedule" },
   679→   679→      })
   680→   680→      break
   681→   681→    case "archive":
   682→   682→      await db.campaign.update({
   683→   683→        where: { id: campaignId },
   684→   684→        data: { state: "archived", sendingNow: false, nextAction: "Archived" },
   685→   685→      })
   686→   686→      break
   687→   687→    case "unarchive":
   688→   688→      await db.campaign.update({
   689→   689→        where: { id: campaignId },
   690→   690→        data: { state: "paused", nextAction: "Paused — press Resume to continue" },
   691→   691→      })
   692→   692→      break
   693→   693→    default:
   694→   694→      return { ok: false, error: `Unknown action: ${action}` }
   695→   695→  }
   696→   696→  return { ok: true }
   697→   697→}
