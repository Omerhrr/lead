    27→function rand(min: number, max: number) {
    28→  return min + Math.random() * (max - min)

    30→function randInt(min: number, max: number) {
    31→  return Math.floor(rand(min, max + 1))

    33→function pickWeighted<T>(entries: Array<{ p: number; outcome: T }>): T {
    34→  const roll = Math.random()
    35→  let acc = 0
    36→  for (const e of entries) {
    37→    acc += e.p
    38→    if (roll <= acc) return e.outcome
    39→  }
    40→  return entries[entries.length - 1].outcome
    41→}

    51→function tokenMatch(haystack: string, needles: string[]): boolean {
    52→  const h = haystack.toLowerCase()
    53→  return needles.some((n) => {
    54→    const t = n.toLowerCase().trim()
    55→    return t.length > 2 && (h.includes(t) || t.includes(h))
    56→  })
    57→}

    59→export async function generateLeadPool(campaignId: string) {
    60→  const campaign = await db.campaign.findUnique({
    61→    where: { id: campaignId },
    62→    include: { project: true },
    63→  })
    64→  if (!campaign) throw new Error("Campaign not found")
    65→
    66→  const filters: LeadFilters = campaign.filters
    67→    ? JSON.parse(campaign.filters)
    68→    : { industries: [], titles: [], geography: [], companySize: [], keywords: [] }
    69→
    70→  const user = await db.user.findUnique({
    71→    where: { id: campaign.project.userId },
    72→    include: { suppress: true },
    73→  })
    74→  const suppressedEmails = new Set(
    75→    (user?.suppress ?? []).filter((s) => s.type === "person").map((s) => s.value.toLowerCase())
    76→  )
    77→  const suppressedCompanies = new Set(
    78→    (user?.suppress ?? [])
    79→      .filter((s) => s.type === "company" || s.type === "domain")
    80→      .map((s) => s.value.toLowerCase())
    81→  )
    82→
    83→  const rows = await db.dbPerson.findMany({ include: { company: true } })
    84→
    85→  const scoreRow = (p: (typeof rows)[number]) => {
    86→    let industryMatch = tokenMatch(p.company.industry, filters.industries)
    87→    let kwMatch = false
    88→    if (!industryMatch && filters.keywords.length) {
    89→      kwMatch = tokenMatch(p.company.industry, filters.keywords) || tokenMatch(p.company.name, filters.keywords)
    90→    }
    91→    const geoMatch = filters.geography.length === 0 || tokenMatch(p.company.country, filters.geography)
    92→    const sizeMatch = filters.companySize.length === 0 || filters.companySize.includes(p.company.size)
    93→    const titleMatch = filters.titles.length === 0 || tokenMatch(p.title, filters.titles)
    94→    const score = Math.min(
    95→      96,
    96→      38 + p.seniority * 9 + (industryMatch ? 10 : kwMatch ? 6 : 0) + (geoMatch ? 6 : 0) + (sizeMatch ? 5 : 0) + (titleMatch ? 6 : 0)
    97→    )
    98→    return { p, industryMatch, kwMatch, geoMatch, sizeMatch, titleMatch, score }
    99→  }
   100→
   101→  const strict = rows.map(scoreRow).filter(
   102→    (r) => r.industryMatch && r.titleMatch && r.geoMatch && r.sizeMatch
   103→  )
   104→  const relaxed = rows.map(scoreRow).filter(
   105→    (r) => (r.industryMatch || r.kwMatch) && r.titleMatch && r.geoMatch
   106→  )
   107→  const fallback = rows.map(scoreRow).filter((r) => r.titleMatch && r.score >= 55)
   108→
   109→  const pool =
   110→    strict.length >= 24 ? strict : relaxed.length >= 24 ? relaxed : [...relaxed, ...fallback]
   111→
   112→  const seen = new Set<string>()
   113→  const chosen = pool
   114→    .sort((a, b) => b.score - a.score)
   115→    .filter((r) => {
   116→      if (seen.has(r.p.email)) return false
   117→      if (suppressedEmails.has(r.p.email.toLowerCase())) return false
   118→      if (suppressedCompanies.has(r.p.company.domain.toLowerCase()) || suppressedCompanies.has(r.p.company.name.toLowerCase())) return false
   119→      seen.add(r.p.email)
   120→      return true
   121→    })
   122→    .slice(0, 32)
   123→
   124→  await db.lead.deleteMany({ where: { campaignId, status: "pool" } })
   125→
   126→  await db.$transaction(
   127→    chosen.map((r) =>
   128→      db.lead.create({
   129→        data: {
   130→          campaignId,
   131→          name: r.p.name,
   132→          title: r.p.title,
   133→          company: r.p.company.name,
   134→          companyDomain: r.p.company.domain,
   135→          industry: r.p.company.industry,
   136→          companySize: r.p.company.size,
   137→          country: r.p.company.country,
   138→          city: r.p.company.city,
   139→          email: r.p.email,
   140→          linkedinUrl: r.p.linkedin,
   141→          score: r.score,
   142→          reasons: r.p.hook,
   143→          status: "pool",
   144→        },
   145→      })
   146→    )
   147→  )
   148→
   149→  await db.campaign.update({
   150→    where: { id: campaignId },
   151→    data: {
   152→      state: "pool_ready",
   153→      nextAction: "Review the lead pool and approve leads to unlock sending",
   154→    },
   155→  })
   156→
   157→  return chosen.length
   158→}

   172→async function charge(accountId: string, costUsd: number, description: string, opts: { emails: number; linkedin: number }) {
   173→  const account = await db.billingAccount.findUnique({ where: { id: String(accountId) } })
   174→  if (!account) return
   175→  await db.billingAccount.update({
   176→    where: { id: account.id },
   177→    data: {
   178→      prepaidBalanceUsd: account.prepaidBalanceUsd - costUsd,
   179→      spentTodayUsd: account.spentTodayUsd + costUsd,
   180→      spentThisMonthUsd: account.spentThisMonthUsd + costUsd,
   181→    },
   182→  })
   183→  await db.creditTransaction.create({
   184→    data: {
   185→      accountId: account.id,
   186→      type: "spend",
   187→      amountUsd: -costUsd,
   188→      balanceAfter: account.prepaidBalanceUsd - costUsd,
   189→      description,
   190→    },
   191→  })
   192→  const date = todayStr()
   193→  const existing = await db.usageDaily.findUnique({
   194→    where: { accountId_date: { accountId: account.id, date } },
   195→  })
   196→  if (existing) {
   197→    await db.usageDaily.update({
   198→      where: { id: existing.id },
   199→      data: {
   200→        emailsSent: existing.emailsSent + opts.emails,
   201→        linkedinTouches: existing.linkedinTouches + opts.linkedin,
   202→        spendUsd: existing.spendUsd + costUsd,
   203→      },
   204→    })
   205→  } else {
   206→    await db.usageDaily.create({
   207→      data: {
   208→        accountId: account.id,
   209→        date,
   210→        emailsSent: opts.emails,
   211→        linkedinTouches: opts.linkedin,
   212→        spendUsd: costUsd,
   213→      },
   214→    })
   215→  }
   216→}

   218→async function ensureDailyReset(accountId: string) {
   219→  const account = await db.billingAccount.findUnique({ where: { id: accountId } })
   220→  if (!account) return null
   221→  const today = todayStr()
   222→  if (account.spentTodayDate !== today) {

   224→      where: { id: account.id },
   225→      data: { spentTodayDate: today, spentTodayUsd: 0 },

   227→  }

   275→export async function tick(userId: string) {
   276→  const account = await ensureDailyReset(
   277→    (await db.billingAccount.findUnique({ where: { userId } }))?.id ?? ""
   278→  )
   279→  if (!account) return { ok: false }
   280→
   281→  const projects = await db.project.findMany({ where: { userId } })
   282→  const projectIds = projects.map((p) => p.id)
   283→  const activeCampaigns = await db.campaign.findMany({
   284→    where: { projectId: { in: projectIds }, state: "sending", sendingNow: true },
   285→    include: { project: true },
   286→  })
   287→
   288→  const pulse = { sent: 0, replies: 0, booked: 0, spent: 0 }
   289→
   290→  for (const campaign of activeCampaigns) {
   291→    await advanceCampaign(campaign, account, pulse)
   292→  }
   293→  return { ok: true, ...pulse }
   294→}

   303→async function advanceCampaign(
   304→  campaign: { id: string; projectId: string; maxTouches: number; delayDays: number; autopilotEnabled: boolean; autoReplyEnabled: boolean; autoReplyDelayMinutes: number; copySubject: string | null; copyBody: string | null; dailyBudgetUsd: number; name: string; offer: string; customerProblem: string; project: { senderName: string; companyName: string } },
   305→  account: { id: string; prepaidBalanceUsd: number; spentTodayUsd: number; spentThisMonthUsd: number; monthlyCapUsd: number; dailyBudgetUsd: number; emailPriceUsd: number; linkedinSeats: number; linkedinSeatBilledMonth: string },
   306→  pulse: { sent: number; replies: number; booked: number; spent: number }
   307→) {
   308→  const now = new Date()
   309→
   310→  // 1) inbound replies due
   311→  const repliesDue = await db.lead.findMany({
   312→    where: {
   313→      campaignId: campaign.id,
   314→      replyDueAt: { lte: now },
   315→      status: "approved",
   316→      OR: [{ convStatus: null }, { convStatus: { in: ["negotiating", "question"] } }],
   317→    },
   318→  })
   319→  for (const lead of repliesDue) {
   320→    await processInbound(campaign, lead, account)
   321→    pulse.replies++
   322→  }
   323→
   324→  // 2) autopilot actions due
   325→  if (campaign.autopilotEnabled && campaign.autoReplyEnabled) {
   326→    const agentDue = await db.lead.findMany({
   327→      where: { campaignId: campaign.id, agentDueAt: { lte: now }, agentStage: { not: null }, convStatus: { in: ["interested", "negotiating", "question", "propose"] } },
   328→    })
   329→    for (const lead of agentDue.slice(0, 3)) {
   330→      await processAutopilot(campaign, lead)
   331→    }
   332→  }
   333→
   334→  // 3) budget guards
   335→  const capacity = await campaignCapacity(campaign.id, campaign.projectId)
   336→  const quota = Math.max(2, Math.ceil(capacity / 12))
   337→  const perSendCost = account.emailPriceUsd || EMAIL_PRICE_USD
   338→  const balanceOk = account.prepaidBalanceUsd >= perSendCost
   339→  const dailyOk = account.spentTodayUsd + perSendCost <= Math.min(account.dailyBudgetUsd, campaign.dailyBudgetUsd)
   340→  const monthlyOk = account.spentThisMonthUsd < account.monthlyCapUsd
   341→
   342→  if (!balanceOk) {
   343→    await db.campaign.update({
   344→      where: { id: campaign.id },
   345→      data: { nextAction: "Out of credits — top up billing to resume outreach" },
   346→    })
   347→    return
   348→  }
   349→  if (!monthlyOk) {
   350→    await db.campaign.update({
   351→      where: { id: campaign.id },
   352→      data: { nextAction: "Monthly cap reached — raise the cap in billing" },
   353→    })
   354→    return
   355→  }
   356→  if (!dailyOk) {
   357→    await db.campaign.update({
   358→      where: { id: campaign.id },
   359→      data: { nextAction: `Daily budget reached (${campaign.dailyBudgetUsd} USD) — resumes next cycle` },
   360→    })
   361→    return
   362→  }
   363→  if (capacity === 0) {
   364→    await db.campaign.update({
   365→      where: { id: campaign.id },
   366→      data: { nextAction: "No active sending domains — buy domains in the Domains tab" },
   367→    })
   368→    return
   369→  }
   370→
   371→  // 4) sends (first touches + follow-ups)
   372→  const eligible = await db.lead.findMany({
   373→    where: {
   374→      campaignId: campaign.id,
   375→      status: "approved",
   376→      convStatus: null,
   377→      OR: [
   378→        { touchCount: 0 },
   379→        { nextTouchAt: { lte: now }, touchCount: { lt: campaign.maxTouches } },
   380→      ],
   381→    },
   382→    orderBy: [{ touchCount: "asc" }, { score: "desc" }],
   383→    take: quota,
   384→  })
   385→
   386→  let emails = 0
   387→  let linkedin = 0
   388→  for (const lead of eligible) {
   389→    const isFollowup = lead.touchCount > 0
   390→    const seatDue = account.linkedinSeatBilledMonth !== monthStr()
   391→    const seatCost = account.linkedinSeats * LINKEDIN_SEAT_PRICE_USD
   392→    const seatAffordable = !seatDue || account.prepaidBalanceUsd >= seatCost + perSendCost
   393→    const useLinkedin = !!lead.linkedinUrl && lead.touchCount >= 1 && seatAffordable && Math.random() < 0.3
   394→    const channel = useLinkedin ? "linkedin" : "email"
   395→    const touchIndex = lead.touchCount + 1
   396→
   397→    const subject = isFollowup
   398→      ? touchIndex === 2
   399→        ? `Re: ${(campaign.copySubject || "Quick question").replace(/^Re:\s*/i, "")}`
   400→        : `Following up (last one, promise)`
   401→      : (campaign.copySubject || "Quick question")
   402→    const bodyTpl = campaign.copyBody || `Hi {{first_name}}, quick idea for {{company}}.`
   403→    let body = fillCopy(bodyTpl, lead, campaign.project.senderName)
   404→    if (isFollowup) {
   405→      body = `Hi ${lead.name.split(" ")[0]}, just floating this back to the top of your inbox — happy to share how similar teams fixed "${lead.reasons || "this"}" with us.\n\nWorth 15 minutes?`
   406→      if (touchIndex >= campaign.maxTouches) body += `\n\nI'll close the loop on my end either way — good luck with ${lead.company}!`
   407→    }
   408→
   409→    await db.message.create({
   410→      data: {
   411→        campaignId: campaign.id,
   412→        leadId: lead.id,
   413→        direction: "out",
   414→        channel,
   415→        kind: "touch",
   416→        touchIndex,
   417→        subject: channel === "email" ? subject : null,
   418→        body,
   419→        status: Math.random() < 0.03 ? "bounced" : "sent",
   420→      },
   421→    })
   422→
   423→    const nextTouchAt =
   424→      touchIndex < campaign.maxTouches
   425→        ? new Date(now.getTime() + campaign.delayDays * FOLLOWUP_MS_PER_DAY)
   426→        : null
   427→
   428→    const willReply =
   429→      lead.replyDueAt === null &&
   430→      Math.random() < (REPLY_CHANCE_BY_TOUCH[Math.min(touchIndex - 1, REPLY_CHANCE_BY_TOUCH.length - 1)] ?? 0.08)
   431→
   432→    await db.lead.update({
   433→      where: { id: lead.id },
   434→      data: {
   435→        touchCount: touchIndex,
   436→        nextTouchAt,
   437→        replyDueAt: willReply ? new Date(now.getTime() + rand(15000, 55000)) : lead.replyDueAt,
   438→        replyChannel: willReply ? "email" : lead.replyChannel,
   439→      },
   440→    })
   441→
   442→    if (channel === "email") emails++
   443→    else linkedin++
   444→  }
   445→
   446→  if (emails + linkedin > 0) {
   447→    const cost = emails * perSendCost + linkedin * LINKEDIN_TOUCH_PRICE_USD
   448→    let description = `${campaign.name}: ${emails} email${emails === 1 ? "" : "s"} sent`
   449→    if (linkedin > 0) description += `, ${linkedin} LinkedIn touch${linkedin === 1 ? "" : "es"}`
   450→    await charge(account.id, cost, description, { emails, linkedin })
   451→    pulse.sent += emails
   452→    pulse.spent += cost
   453→
   454→    // LinkedIn seat billing: once per month on first touch — never overdraw:
   455→    // if the balance can't cover the seats, LinkedIn touches fall back to email.
   456→    if (linkedin > 0 && account.linkedinSeatBilledMonth !== monthStr()) {
   457→      const seatCost = account.linkedinSeats * LINKEDIN_SEAT_PRICE_USD
   458→      const fresh = await db.billingAccount.findUnique({ where: { id: account.id } })
   459→      if (fresh && fresh.prepaidBalanceUsd >= seatCost) {
   460→        await db.billingAccount.update({
   461→          where: { id: account.id },
   462→          data: {
   463→            prepaidBalanceUsd: fresh.prepaidBalanceUsd - seatCost,
   464→            linkedinSeatBilledMonth: monthStr(),
   465→          },
   466→        })
   467→        await db.creditTransaction.create({
   468→          data: {
   469→            accountId: account.id,
   470→            type: "linkedin_seats",
   471→            amountUsd: -seatCost,
   472→            balanceAfter: fresh.prepaidBalanceUsd - seatCost,
   473→            description: `LinkedIn seats — ${account.linkedinSeats} seat${account.linkedinSeats === 1 ? "" : "s"} (${monthStr()})`,
   474→          },
   475→        })
   476→      } else if (fresh) {
   477→        await db.campaign.update({
   478→          where: { id: campaign.id },
   479→          data: {
   480→            nextAction: `LinkedIn seats not charged — balance below ${seatCost} USD; touches sent via email instead`,
   481→          },
   482→        })
   483→      }
   484→    }
   485→  }
   486→
   487→  // 5) completion check
   488→  const approvedLeads = await db.lead.findMany({ where: { campaignId: campaign.id, status: "approved" } })
   489→  const done = approvedLeads.every(
   490→    (l) =>
   491→      ["booked", "closed", "unsubscribed", "active"].includes(l.convStatus ?? "") ||
   492→      (l.convStatus === null && l.touchCount >= campaign.maxTouches && !l.nextTouchAt)
   493→  )
   494→  if (approvedLeads.length > 0 && done) {
   495→    await db.campaign.update({
   496→      where: { id: campaign.id },
   497→      data: { state: "completed", sendingNow: false, nextAction: "Campaign completed — review hot leads in the inbox" },
   498→    })
   499→  } else {
   500→    await db.campaign.update({
   501→      where: { id: campaign.id },
   502→      data: { nextAction: "Sending — sequences running on schedule" },
   503→    })
   504→  }
   505→}

   509→async function processInbound(
   510→  campaign: { id: string; autopilotEnabled: boolean; autoReplyEnabled: boolean; autoReplyDelayMinutes: number },
   511→  lead: { id: string; name: string; company: string; convStatus: string | null; replyChannel: string | null },
   512→  account: { id: string }
   513→) {
   514→  const now = new Date()
   515→  let outcome: string
   516→  if (lead.convStatus === "negotiating") {
   517→    outcome = "negotiating"
   518→  } else if (lead.convStatus === "question") {
   519→    outcome = Math.random() < 0.5 ? "interested" : "not_interested"
   520→  } else {
   521→    outcome = pickWeighted(FIRST_REPLY_OUTCOMES)
   522→  }
   523→
   524→  const pool = INBOUND_TEXT[outcome] ?? INBOUND_TEXT.not_interested
   525→  const text = pool[randInt(0, pool.length - 1)]
   526→
   527→  await db.message.create({
   528→    data: {
   529→      campaignId: campaign.id,
   530→      leadId: lead.id,
   531→      direction: "in",
   532→      channel: lead.replyChannel || "email",
   533→      kind: "reply",
   534→      body: text,
   535→      status: "sent",
   536→    },
   537→  })
   538→
   539→  const hot = outcome === "interested" || outcome === "negotiating"
   540→  const updates: Record<string, unknown> = { replyDueAt: null, sentiment: outcome }
   541→  switch (outcome) {
   542→    case "interested":
   543→      updates.convStatus = "interested"
   544→      updates.isHot = true
   545→      if (campaign.autopilotEnabled && campaign.autoReplyEnabled) {
   546→        updates.agentDueAt = new Date(now.getTime() + Math.max(10, campaign.autoReplyDelayMinutes) * 1000)
   547→        updates.agentStage = "propose_meeting"
   548→      } else {
   549→        updates.convStatus = "active"
   550→      }
   551→      break
   552→    case "question":
   553→      updates.convStatus = "question"
   554→      if (campaign.autopilotEnabled && campaign.autoReplyEnabled) {
   555→        updates.agentDueAt = new Date(now.getTime() + Math.max(10, campaign.autoReplyDelayMinutes) * 1000)
   556→        updates.agentStage = "answer"
   557→      } else {
   558→        updates.convStatus = "active"
   559→      }
   560→      break
   561→    case "not_interested":
   562→      updates.convStatus = "closed"
   563→      break
   564→    case "unsubscribe":
   565→      updates.convStatus = "unsubscribed"
   566→      updates.status = "suppressed"
   567→      break
   568→    case "negotiating":
   569→      // lead accepted the meeting
   570→      updates.convStatus = "negotiating"
   571→      updates.isHot = true
   572→      updates.agentDueAt = new Date(now.getTime() + Math.max(10, campaign.autoReplyDelayMinutes) * 1000)
   573→      updates.agentStage = "confirm"
   574→      updates.meetingAt = new Date(now.getTime() + MEETING_LEAD_DAYS[randInt(0, 3)] * 24 * 3600 * 1000)
   575→      break
   576→    case "booked_confirm":
   577→      updates.convStatus = "booked"
   578→      break
   579→  }
   580→
   581→  await db.lead.update({ where: { id: lead.id }, data: updates })
   582→  await noteReplyEvent(account.id, hot)
   583→}

   587→async function processAutopilot(
   588→  campaign: { id: string; autoReplyDelayMinutes: number; offer: string; customerProblem: string; project: { companyName: string; senderName: string } },
   589→  lead: { id: string; name: string; company: string; convStatus: string | null; agentStage: string | null; meetingAt: Date | null; replyChannel: string | null }
   590→) {
   591→  const now = new Date()
   592→  const stage = lead.agentStage as "propose_meeting" | "confirm" | "answer"
   593→  const companyName = campaign.project.companyName || "our team"
   594→
   595→  // find the last inbound message to answer
   596→  const lastIn = await db.message.findFirst({
   597→    where: { leadId: lead.id, direction: "in" },
   598→    orderBy: { createdAt: "desc" },
   599→  })
   600→
   601→  const slot = `${["Monday", "Tuesday", "Wednesday", "Thursday", "Friday"][randInt(0, 4)]}, ${["10:00", "11:30", "14:00", "15:30"][randInt(0, 3)]}`
   602→  const text = await autopilotReply({
   603→    companyName,
   604→    offer: campaign.offer,
   605→    customerProblem: campaign.customerProblem,
   606→    leadName: lead.name,
   607→    leadCompany: lead.company,
   608→    leadMessage: lastIn?.body ?? "Interested in learning more.",
   609→    stage,
   610→    slot,
   611→  })
   612→
   613→  await db.message.create({
   614→    data: {
   615→      campaignId: campaign.id,
   616→      leadId: lead.id,
   617→      direction: "out",
   618→      channel: lead.replyChannel || "email",
   619→      kind: "autopilot",
   620→      touchIndex: 0,
   621→      subject: null,
   622→      body: text,
   623→      status: "sent",
   624→    },
   625→  })
   626→
   627→  const updates: Record<string, unknown> = { agentDueAt: null, agentStage: null }
   628→  if (stage === "propose_meeting") {
   629→    updates.convStatus = "negotiating"
   630→    updates.replyDueAt = new Date(now.getTime() + rand(15000, 40000))
   631→    updates.replyChannel = "email"
   632→  } else if (stage === "confirm") {
   633→    updates.convStatus = "booked"
   634→    updates.meetingAt =
   635→      lead.meetingAt ?? new Date(now.getTime() + MEETING_LEAD_DAYS[randInt(0, 3)] * 24 * 3600 * 1000)
   636→  } else if (stage === "answer") {
   637→    // lead decides after the answer
   638→    updates.convStatus = "question"
   639→    updates.replyDueAt = new Date(now.getTime() + rand(15000, 40000))
   640→    updates.replyChannel = "email"
   641→  }
   642→  await db.lead.update({ where: { id: lead.id }, data: updates })
   643→}

   647→export async function campaignAction(campaignId: string, action: string) {
   648→  const campaign = await db.campaign.findUnique({ where: { id: campaignId } })
   649→  if (!campaign) throw new Error("Campaign not found")
   650→
   651→  switch (action) {
   652→    case "approve":
   653→      await db.campaign.update({
   654→        where: { id: campaignId },
   655→        data: { state: "approved", nextAction: "Press Start to begin outreach" },
   656→      })
   657→      break
   658→    case "start": {
   659→      const owned = await db.sendingDomain.count({ where: { projectId: campaign.projectId, status: "active" } })
   660→      if (owned === 0) {
   661→        return { ok: false, error: "No active sending domains. Buy domains in the Domains tab first." }
   662→      }
   663→      await db.campaign.update({
   664→        where: { id: campaignId },
   665→        data: { state: "sending", sendingNow: true, nextAction: "Sending — sequences running on schedule" },
   666→      })
   667→      break
   668→    }
   669→    case "stop":
   670→      await db.campaign.update({
   671→        where: { id: campaignId },
   672→        data: { sendingNow: false, state: "paused", nextAction: "Paused — press Resume to continue" },
   673→      })
   674→      break
   675→    case "resume":
   676→      await db.campaign.update({
   677→        where: { id: campaignId },
   678→        data: { sendingNow: true, state: "sending", nextAction: "Sending — sequences running on schedule" },
   679→      })
   680→      break
   681→    case "archive":
   682→      await db.campaign.update({
   683→        where: { id: campaignId },
   684→        data: { state: "archived", sendingNow: false, nextAction: "Archived" },
   685→      })
   686→      break
   687→    case "unarchive":
   688→      await db.campaign.update({
   689→        where: { id: campaignId },
   690→        data: { state: "paused", nextAction: "Paused — press Resume to continue" },
   691→      })
   692→      break
   693→    default:
   694→      return { ok: false, error: `Unknown action: ${action}` }
   695→  }
   696→  return { ok: true }
   697→}
